# pcmod export window on a small mod folder (C:/games/MyMod: the PovertyCaster pcmod fixture files)
wait 30
pcmod open
pcmod addfolder C:/games/MyMod/data
pcmod addfolder C:/games/MyMod/grp
pcmod set id my-mod
pcmod set version 1.2.3
pcmod set name My MBAACC mod
pcmod set exe 6d1415ca9573100e86a779ac2f81e9bedd322664e3daeae0229a67d13720310a
pcmod set priority 100
pcmod set out C:/games/MyMod/out
pcmod status
wait 10
capture C:/dev/hantei-chan/work/ab/pcmod_export_window.png
pcmod export
wait 10
capture C:/dev/hantei-chan/work/ab/pcmod_export_done.png
quit
